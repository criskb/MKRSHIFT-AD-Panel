Task 2.1 — Add TransitionPass
Create file: src/postfx/TransitionPass.js
import * as THREE from "three";

export const TransitionShader = {
  uniforms: {
    tFrom: { value: null },
    tTo: { value: null },
    progress: { value: 0 },
    time: { value: 0 },
    softness: { value: 0.2 }, // edge softness
    noiseScale: { value: 2.0 },
    rgbSplit: { value: 0.001 }, // subtle
  },
  vertexShader: `
    varying vec2 vUv;
    void main(){
      vUv = uv;
      gl_Position = vec4(position.xy, 0.0, 1.0);
    }
  `,
  fragmentShader: `
    precision highp float;
    varying vec2 vUv;
    uniform sampler2D tFrom;
    uniform sampler2D tTo;
    uniform float progress;
    uniform float time;
    uniform float softness;
    uniform float noiseScale;
    uniform float rgbSplit;

    // cheap hash noise
    float hash(vec2 p){
      p = fract(p * vec2(123.34, 345.45));
      p += dot(p, p + 34.345);
      return fract(p.x * p.y);
    }

    void main(){
      vec2 uv = vUv;

      float n = hash(uv * noiseScale + time * 0.1);
      float m = smoothstep(progress - softness, progress + softness, n);

      // subtle chromatic split near the edge
      float edge = abs(n - progress);
      float split = rgbSplit * smoothstep(0.2, 0.0, edge);

      vec4 a = texture2D(tFrom, uv);
      vec4 bR = texture2D(tTo, uv + vec2(split, 0.0));
      vec4 bG = texture2D(tTo, uv);
      vec4 bB = texture2D(tTo, uv - vec2(split, 0.0));
      vec4 b = vec4(bR.r, bG.g, bB.b, bG.a);

      gl_FragColor = mix(a, b, m);
    }
  `,
};

export function createTransitionMaterial() {
  return new THREE.ShaderMaterial({
    uniforms: THREE.UniformsUtils.clone(TransitionShader.uniforms),
    vertexShader: TransitionShader.vertexShader,
    fragmentShader: TransitionShader.fragmentShader,
    depthTest: false,
    depthWrite: false,
  });
}

// --- Add uniforms:
uniform float uVignette;   // 0..1
uniform float uGrain;      // 0..1
uniform float uSharpen;    // 0..1
uniform float uChromAb;    // 0..0.01

float rand(vec2 co){
  return fract(sin(dot(co.xy ,vec2(12.9898,78.233))) * 43758.5453);
}

vec3 sharpen(sampler2D tex, vec2 uv, vec2 px, float amount){
  vec3 c = texture2D(tex, uv).rgb;
  vec3 n = texture2D(tex, uv + vec2(0.0, px.y)).rgb;
  vec3 s = texture2D(tex, uv - vec2(0.0, px.y)).rgb;
  vec3 e = texture2D(tex, uv + vec2(px.x, 0.0)).rgb;
  vec3 w = texture2D(tex, uv - vec2(px.x, 0.0)).rgb;
  vec3 edge = (n + s + e + w) * 0.25 - c;
  return c - edge * amount;
}

// --- In main():
vec2 px = vec2(1.0/uResolution.x, 1.0/uResolution.y);

// Chromatic aberration
vec2 ca = (vUv - 0.5) * uChromAb;
vec3 col;
col.r = texture2D(uTex, vUv + ca).r;
col.g = texture2D(uTex, vUv).g;
col.b = texture2D(uTex, vUv - ca).b;

// Sharpen
col = mix(col, sharpen(uTex, vUv, px, 1.0), uSharpen);

// Vignette
float d = distance(vUv, vec2(0.5));
float vig = smoothstep(0.8, 0.2, d);
col *= mix(1.0, vig, uVignette);

// Grain
float g = rand(vUv * uTime) - 0.5;
col += g * uGrain * 0.08;

gl_FragColor = vec4(col, 1.0);

Below is a Codex-ready task list to fix MP4/WebM video + GIFs stuck on first frame and make playback solid for a kiosk/digital-signage setup.
The two most common causes are:
	1	Autoplay restrictions: video.play() is getting blocked (must be muted + usually needs a user gesture once). play()returns a Promise and will reject when blocked. (MDN Web Docs)
	2	Wrong texture type / no updates: using THREE.Texture(video) (static) instead of THREE.VideoTexture(video), or using createImageBitmap() on GIF/video (often yields only first frame).

import * as THREE from "three";

export async function createVideoSourceFromFile(file) {
  const url = URL.createObjectURL(file);

  const video = document.createElement("video");
  video.src = url;

  // kiosk-friendly defaults
  video.muted = true;             // muted autoplay is generally allowed
  video.loop = true;
  video.playsInline = true;       // important for iOS/Safari inline playback
  video.autoplay = true;
  video.preload = "auto";

  // For Safari/iOS quirks:
  video.setAttribute("playsinline", "");
  video.setAttribute("webkit-playsinline", "");

  // Wait until the browser has enough data to start
  await new Promise((resolve, reject) => {
    const onReady = () => resolve();
    const onErr = (e) => reject(e);
    video.addEventListener("canplay", onReady, { once: true });
    video.addEventListener("error", onErr, { once: true });
    video.load();
  });

  // Try to play; if blocked, you MUST ask for a click/tap once
  try {
    await video.play(); // returns a Promise; rejects when autoplay denied
  } catch (err) {
    // Show “Click to start” overlay once; then call video.play() again.
    // Autoplay failures are expected on many setups. :contentReference[oaicite:1]{index=1}
    throw new Error("AUTOPLAY_BLOCKED");
  }

  const texture = new THREE.VideoTexture(video);
  texture.colorSpace = THREE.SRGBColorSpace;
  texture.minFilter = THREE.LinearFilter;
  texture.magFilter = THREE.LinearFilter;
  texture.generateMipmaps = false;

  return {
    kind: "video",
    video,
    texture,
    url,
    dispose() {
      try { video.pause(); } catch {}
      video.removeAttribute("src");
      video.load();
      URL.revokeObjectURL(url);
      texture.dispose();
    },
  };
}

✅ Task List: Fix GIF playback (animated, not first-frame only)
Choose one of these approaches:
Option A (fast patch): Keep <img> animated and force texture updates
This is the simplest fix if you already treat GIFs like images.
import * as THREE from "three";

export async function createGifSourceQuick(file) {
  const url = URL.createObjectURL(file);

  const img = new Image();
  img.decoding = "async";
  img.src = url;

  await img.decode(); // decodes first frame; animation continues in <img>

  const texture = new THREE.Texture(img);
  texture.colorSpace = THREE.SRGBColorSpace;
  texture.minFilter = THREE.LinearFilter;
  texture.magFilter = THREE.LinearFilter;
  texture.generateMipmaps = false;
  texture.needsUpdate = true;

  // Critical: GIF frames will NOT push updates into WebGL automatically.
  // You must mark needsUpdate repeatedly.
  let alive = true;
  function update() {
    if (!alive) return;
    texture.needsUpdate = true;
  }

  return {
    kind: "gif",
    img,
    texture,
    update, // call this in your render loop (maybe throttle to 30fps)
    dispose() {
      alive = false;
      URL.revokeObjectURL(url);
      texture.dispose();
    },
  };
}
In your render loop:
// throttle GIF updates to reduce GPU uploads
let lastGifUpdate = 0;
function tick(t) {
  if (gifSource && (t - lastGifUpdate) > 33) { // ~30fps
    gifSource.update();
    lastGifUpdate = t;
  }
  renderer.render(scene, camera);
  requestAnimationFrame(tick);
}

Option B (best): Decode GIF frames and drive them yourself (smooth + efficient)
If you want “proper signage-grade” GIF handling, decode frames and blit onto a canvas, then use THREE.CanvasTexture.
Use gifuct-js (small, reliable). Create:
import * as THREE from "three";
import { parseGIF, decompressFrames } from "gifuct-js";

export async function createGifSourceDecoded(file) {
  const buf = await file.arrayBuffer();
  const gif = parseGIF(buf);
  const frames = decompressFrames(gif, true); // gives RGBA patches + delays

  const width = gif.lsd.width;
  const height = gif.lsd.height;

  const canvas = document.createElement("canvas");
  canvas.width = width;
  canvas.height = height;
  const ctx = canvas.getContext("2d", { willReadFrequently: true });

  const imageData = ctx.createImageData(width, height);

  // Start with blank
  ctx.clearRect(0, 0, width, height);

  const texture = new THREE.CanvasTexture(canvas);
  texture.colorSpace = THREE.SRGBColorSpace;
  texture.minFilter = THREE.LinearFilter;
  texture.magFilter = THREE.LinearFilter;
  texture.generateMipmaps = false;

  let i = 0;
  let acc = 0;
  let alive = true;

  function drawFrame(frame) {
    // frame.patch is Uint8ClampedArray RGBA for the frame area
    // frame.dims: { width, height, left, top }
    const { width: fw, height: fh, left, top } = frame.dims;

    // Put patch into ImageData at correct offset
    // Easiest: create temp ImageData for patch then putImageData
    const patch = new ImageData(frame.patch, fw, fh);
    ctx.putImageData(patch, left, top);

    texture.needsUpdate = true;
  }

  function update(dtMs) {
    if (!alive) return;
    acc += dtMs;

    // delay is in hundredths of a second in many GIFs; gifuct gives ms-ish delay per frame
    const delay = Math.max(20, frames[i].delay || 100); // clamp for “0 delay” gifs
    if (acc >= delay) {
      acc = acc % delay;
      drawFrame(frames[i]);
      i = (i + 1) % frames.length;
    }
  }

  return {
    kind: "gif",
    texture,
    update, // call update(deltaMs)
    dispose() {
      alive = false;
      texture.dispose();
    },
  };
}
In your animation loop (delta time based):
let lastT = performance.now();
function tick(t) {
  const dt = t - lastT;
  lastT = t;

  if (gifDecoded) gifDecoded.update(dt);

  renderer.render(scene, camera);
  requestAnimationFrame(tick);
}

Extra improvements (worth doing)
A) If you sample video frames manually, use requestVideoFrameCallback
If your pipeline needs “update only when a new frame exists” (e.g. custom canvas processing), use:
	•	video.requestVideoFrameCallback(cb) to sync exactly to frame delivery. (MDN Web Docs)
B) Make “Media Source” a unified interface
Have one interface for image/video/gif:
type MediaSource = {
  kind: "image" | "video" | "gif";
  texture: THREE.Texture;
  update?: (dtMs: number) => void;
  dispose: () => void;
}
Then your render loop can simply do:
if (activeMedia?.update) activeMedia.update(dt);
shaderUniforms.uMedia.value = activeMedia.texture;
C) Prevent multiple videos decoding at once
Only keep the active slide’s video playing. Pause others. This prevents CPU spikes and “first frame stuck” edge cases when tab throttling kicks in.

Quick “what to change right now” checklist
	•	 Ensure every video element has: muted=true, playsInline=true, loop=true
	•	 Always await video.play() and handle rejection (show click-to-start) (MDN Web Docs)
	•	 Use THREE.VideoTexture(video) not THREE.Texture(video)
	•	 For GIFs: either (A) set texture.needsUpdate=true repeatedly, or (B) decode + CanvasTexture
	•	 On slide change: pause previous media + dispose object URLs
